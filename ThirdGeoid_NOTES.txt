can you make two javascript functions named "moveEast" and "moveWest"
that cyle through an array if images:
["image1", "image2", "image3"]
at a image element in the html defined by this
<img id="geosta-image">
using two button elements displaying the text " >> " and " << "

<img id="geosta-image" src="image1" alt="Geosta image">

<button onclick="moveWest()"> &lt;&lt; </button>
<button onclick="moveEast()"> &gt;&gt; </button>

<script>
  const images = [
    "image1",
    "image2",
    "image3"
  ];

  let currentImage = 0;

  function moveEast() {
    currentImage = (currentImage + 1) % images.length;

    document.getElementById("geosta-image").src =
      images[currentImage];
  }

  function moveWest() {
    currentImage =
      (currentImage - 1 + images.length) % images.length;

    document.getElementById("geosta-image").src =
      images[currentImage];
  }
</script>


Behavior:

Clicking >> calls moveEast():
image1 → image2 → image3 → image1
Clicking << calls moveWest():
image1 → image3 → image2 → image1
